\q{1}\score{K 4} Solve each equation.
\begin{tasks}(2)
  \task $x + 5 = 12$\onlya{%
    \[\work{x + 5 \bal{{}- 5} = 12 \bal{{}- 5}} \qquad \work{x = 7}\]}
  \task $3x = 18$\onlya{%
    \[\work{\frac{3x}{\bal{3}} = \frac{18}{\bal{3}}} \qquad \work{x = 6}\]}
\end{tasks}

\q{2}\score{A 3} A taxi charges \$4 plus \$2 for each kilometre. How far can you ride for \$20?
\onlyq{\vspace{8\baselineskip}}
\onlya{%
\[\work{4 + 2k = 20} \qquad \work{4 + 2k \bal{{}- 4} = 20 \bal{{}- 4}} \qquad \work{2k = 16} \qquad \work{k = 8}\]
\work{You can ride 8 km.}}

\q[12]{3}\score{T 5} A garden bed is a rectangle. Its width is $w$ metres. Its length is 3 m more than its width.
\partbox{2.4cm}{a}{Write the area of the bed as an expression in $w$. Write it in factored form, then in expanded form.}{\work{$A = w(w + 3)$ \qquad $A = w^2 + 3w$}}
\partbox{3.4cm}{b}{The area of the bed is 40 square metres. Write an equation for the width and solve it. Then say which solution is the width of the bed, and why.}{%
\work{$w^2 + 3w = 40$ \qquad $w^2 + 3w \bal{{}- 40} = 40 \bal{{}- 40}$ \qquad $(w + 8)(w - 5) = 0$ \qquad $w = -8$ or $w = 5$}\par
\work{A width cannot be negative, so the width of the bed is 5 m.}}
\qpart{c}{The length of the bed is 3 m more than its width. Use the width from part b to find the length of the bed, and write the length on the line. \fillin{8 m}}
