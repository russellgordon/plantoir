\q{1}\score{K 4} Solve each equation.
\begin{tasks}(2)
  \task $x + 5 = 12$\onlya{%
    \[\work{x + 5 \bal{{}- 5} = 12 \bal{{}- 5}} \qquad \work{x = 7}\]}
  \task $3x = 18$\onlya{%
    \[\work{\frac{3x}{\bal{3}} = \frac{18}{\bal{3}}} \qquad \work{x = 6}\]}
\end{tasks}

\q{2}\score{A 3} A taxi charges \$4 plus \$2 for each kilometre. How far can you ride for \$20?
\onlyq{\vspace{8\baselineskip}}
\onlya{%
\[\work{4 + 2k = 20} \qquad \work{4 + 2k \bal{{}- 4} = 20 \bal{{}- 4}} \qquad \work{2k = 16} \qquad \work{k = 8}\]
\work{You can ride 8 km.}}
\end{questions}
